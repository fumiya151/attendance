-- テーブル名: income_tax_rates
-- 目的: 所得税の源泉徴収税額表（月額表）のデータを保持する

CREATE TABLE income_tax_rates (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    
    -- 税制適用開始日
    effective_start_date DATE NOT NULL,
    
    -- 社会保険料控除後の給与額 範囲（下限、税込）
    income_from DECIMAL(10, 0) NOT NULL,
    
    -- 社会保険料控除後の給与額 範囲（上限、税込）
    income_to DECIMAL(10, 0) NOT NULL,
    
    -- 扶養親族 0人の場合の税額（甲欄）
    tax_0 DECIMAL(10, 0) NOT NULL,
    
    -- 扶養親族 1人の場合の税額（甲欄）
    tax_1 DECIMAL(10, 0) NOT NULL,
    
    -- 扶養親族 2人の場合の税額（甲欄）
    tax_2 DECIMAL(10, 0) NOT NULL,
    
    -- 税額表の適用区分 (KOU:甲欄, OTSU:乙欄)
    tax_type VARCHAR(4) NOT NULL,

    -- 制約: 適用開始日と給与範囲の重複を避ける
    CONSTRAINT uc_tax_rate UNIQUE (effective_start_date, income_from, tax_type)
);

-- インデックス
CREATE INDEX idx_itr_dates_income ON income_tax_rates (effective_start_date, income_from, income_to);
