-- テーブル名: payroll_settings
-- 目的: 給与計算に必要な共通の料率や基準値（保険料率、割増率など）を管理する

CREATE TABLE payroll_settings (
    -- 主キー: 設定項目を一意に識別するキー（例: OVERTIME_RATE, HEALTH_INSURANCE_RATE）
    setting_key VARCHAR(255) NOT NULL PRIMARY KEY,
    
    -- 設定値: 料率や金額。最大5桁の小数点（例: 0.01250, 1500.00000）
    setting_value DECIMAL(10, 5) NOT NULL,
    
    -- 適用開始日: この設定が有効になる日付
    effective_start_date DATE NOT NULL,
    
    -- 適用終了日: この設定が終了する日付 (NULLの場合は無期限または現在適用中)
    effective_end_date DATE,
    
    -- 説明: この設定が何であるかの簡潔な説明
    description VARCHAR(255),
    
    -- 制約: 適用開始日が適用終了日より前であることを保証（NULLでない場合）
    CONSTRAINT chk_effective_dates CHECK (effective_end_date IS NULL OR effective_start_date <= effective_end_date)
);

-- インデックス: 設定キーと日付による検索を高速化
CREATE INDEX idx_ps_key_dates ON payroll_settings (setting_key, effective_start_date, effective_end_date);
